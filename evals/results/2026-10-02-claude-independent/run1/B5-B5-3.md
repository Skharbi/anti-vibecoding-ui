## Replacement `src/errorReporting.ts`

```ts
// Payloads are built from an allowlist: credentials, user identifiers and
// form input are never sent. Free text is scrubbed because messages/stacks
// can embed values.
const INGEST_URL = "https://logs.example-observability.io/ingest";

// Per page load; correlates events without identifying the user.
const correlationId = crypto.randomUUID();

const SCRUBBERS: Array<[RegExp, string]> = [
  [/\b(?:\d[ -]?){13,19}\b/g, "[redacted-number]"],
  [/[\w.+-]+@[\w-]+\.[\w.-]+/g, "[redacted-email]"],
  [/\bBearer\s+[\w.~+/-]+=*/gi, "Bearer [redacted]"],
  [/\beyJ[\w-]+\.[\w-]+\.[\w-]+\b/g, "[redacted-jwt]"],
  [/([?&#](?:token|access_token|code|key|password|secret)=)[^&#\s]+/gi, "$1[redacted]"],
];

function scrub(text: string | undefined, max = 2000) {
  if (!text) return text;
  return SCRUBBERS.reduce((s, [re, rep]) => s.replace(re, rep), text).slice(0, max);
}

export type ErrorContext = {
  action: string;
  /** Non-sensitive diagnostics only (status, error code). Never form values. */
  meta?: Record<string, string | number | boolean>;
};

export function reportError(err: unknown, context: ErrorContext) {
  const error = err instanceof Error ? err : new Error(String(err));
  try {
    void fetch(INGEST_URL, {
      method: "POST",
      keepalive: true,
      credentials: "omit",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        name: error.name,
        message: scrub(error.message, 500),
        stack: scrub(error.stack),
        action: context.action,
        meta: context.meta,
        route: location.pathname, // no query/fragment
        correlationId,
        ts: new Date().toISOString(),
        release: import.meta.env.VITE_RELEASE,
      }),
    }).catch(() => {});
  } catch {
    // Reporting must never throw into the user's flow.
  }
}
```

## Replacement `src/PaymentForm.tsx`

```tsx
import { reportError } from "./errorReporting";

export class PaymentError extends Error {
  constructor(public status: number) {
    super(`pay failed ${status}`);
    this.name = "PaymentError";
  }
}

export async function submitPayment(values: { cardNumber: string; cvc: string; amount: number }) {
  try {
    const r = await fetch("/api/pay", { method: "POST", body: JSON.stringify(values) });
    if (!r.ok) throw new PaymentError(r.status);
  } catch (e) {
    // Never pass `values`: it holds card number and CVC.
    reportError(e, {
      action: "submitPayment",
      meta: e instanceof PaymentError ? { status: e.status } : { network: true },
    });
    throw e;
  }
}
```
